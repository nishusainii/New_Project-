# =========================================================
# CISAI Squid authentication
# =========================================================

auth_param basic program /usr/lib/squid/basic_ncsa_auth /etc/squid/password
auth_param basic children 5 startup=1 idle=1
auth_param basic realm CISAI Proxy
auth_param basic credentialsttl 2 hours
auth_param basic casesensitive on

acl authenticated_users proxy_auth REQUIRED


# =========================================================
# Approved client IP addresses
# One IP or CIDR network per line
# =========================================================

acl allowed_ips src "/etc/squid/allowed_ips.txt"


# =========================================================
# Safe destination ports
# =========================================================

acl SSL_ports port 443

acl Safe_ports port 80
acl Safe_ports port 443

# Enable only if actually required
# acl Safe_ports port 21
# acl Safe_ports port 1025-65535

acl CONNECT method CONNECT


# =========================================================
# Restrict HTTP methods
# =========================================================

acl Allowed_methods method GET HEAD POST PUT DELETE OPTIONS CONNECT

# Block HTTP TRACE
acl TRACE method TRACE


# =========================================================
# Squid management access
# "manager" and "localhost" are predefined Squid ACLs
# =========================================================

http_access allow localhost manager
http_access deny manager


# =========================================================
# Prevent proxy access to local and link-local destinations
# "to_localhost" and "to_linklocal" are predefined ACLs
# =========================================================

http_access deny to_localhost
http_access deny to_linklocal


# =========================================================
# Listener ports
# Replace 10.31.88.24 with the actual LAN/management
# address clients use to reach this Squid server
# =========================================================

http_port 10.31.1.24:8855 name=jio1
http_port 10.31.1.24:8848 name=jio2
http_port 10.31.1.24:8894 name=airtel

acl jio1 myportname jio1
acl jio2 myportname jio2
acl airtel myportname airtel 

# =========================================================
# Outgoing source address selection
# =========================================================

tcp_outgoing_address 192.168.18.30 jio1
tcp_outgoing_address 192.168.5.30 jio2
tcp_outgoing_address 192.168.21.10 airtel

# =========================================================
# Optional per-client connection limit
# =========================================================

acl too_many_connections maxconn 100


# =========================================================
# Access-control policy
# Order is critical
# =========================================================

http_access deny !Safe_ports
http_access deny CONNECT !SSL_ports
http_access deny TRACE
http_access deny !Allowed_methods
http_access deny too_many_connections

http_access allow allowed_ips authenticated_users

http_access deny all


# =========================================================
# Logging
# Includes listener name, local port and outgoing peer IP
# =========================================================


# =========================================================
# Privacy and identification
# =========================================================

httpd_suppress_version_string on


# =========================================================
# Runtime account and working directory
# =========================================================

cache_effective_user proxy
cache_effective_group proxy
coredump_dir /var/spool/squid

# CISAI custom listener log
logformat cisai_proxy %ts.%03tu client=%>a clientport=%>p duration=%tr result=%Ss/%>Hs bytes=%<st method=%rm destination=%ru user=%un local_port=%lp local_ip=%la
access_log daemon:/var/log/squid/access-cisai.log logformat=cisai_proxy

access_log stdio:/var/log/squid/access-sarg.log squid

#allows the Squid process to use up to 65,536 descriptors.
max_filedescriptors 65536
