for disk in /dev/sd{a..l}; do  

serial="$(lsblk -dn -o SERIAL "$disk" | xargs)"  

wwn="$(lsblk -dn -o WWN "$disk" | xargs)"  

  
slot="$(  

awk -v serial="$serial" '  

/^Drive \/c0\/e252\/s[0-9]+ :/ {  

drive=$2  

gsub(/:$/, "", drive)  

}  

$1 == "SN" && $2 == "=" && $3 == serial {  

print drive  

}  

' "$PERC_DATA"  

)"  

  
role="UNKNOWN"  

  

case "$disk" in  

/dev/sda|/dev/sdb|/dev/sdc|/dev/sdd)  

osd="$(  

ceph-volume lvm list "$disk" 2>/dev/null |  

awk '/^[[:space:]]+osd id/ {print $3; exit}'  

)"  

  
if [[ -n "$osd" ]]; then  

role="CEPH-osd.${osd}"  

else  

role="CEPH-disk-unmapped"  

fi  

;;  

/dev/sde|/dev/sdf|/dev/sdg|/dev/sdh|/dev/sdi|/dev/sdj|/dev/sdk)  

role="ZFS-RAIDZ2"  

;;  

/dev/sdl)  

role="RESERVED"  

;;  

esac  

  
printf "%-8s %-18s %-12s %-20s %-20s\n" \  

"$disk" "$slot" "$serial" "$role" "$wwn"  

done
